cmake_minimum_required(VERSION 3.19)
project(syncshell_caelestia_sdk VERSION 2.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS ShaderTools Core Qml Gui Quick QuickControls2 Concurrent Sql Network DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Qalculate IMPORTED_TARGET libqalculate REQUIRED)
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml")
set(INSTALL_QMLDIR lib/qt6/qml)
set(VERSION 2.2.0)
qt_standard_project_setup(REQUIRES 6.9)
include("${CAELESTIA_SDK}/plugin/cmake/pch.cmake")
include("${CAELESTIA_SDK}/plugin/cmake/qml-module.cmake")
foreach(module Config Plugins Components Internal Models Images Blobs)
  add_subdirectory("${CAELESTIA_SDK}/plugin/src/Caelestia/${module}" "${module}")
endforeach()

set(core_sources)
foreach(file cutils qalculator appdb requests toaster imageanalyser)
  list(APPEND core_sources "${CAELESTIA_SDK}/plugin/src/Caelestia/${file}.cpp")
endforeach()
qml_module(caelestia-core URI Caelestia SOURCES ${core_sources}
 LIBRARIES Qt::Gui Qt::Quick Qt::Concurrent Qt::Sql PkgConfig::Qalculate)
target_compile_definitions(caelestia-core PRIVATE CAELESTIA_VERSION="${VERSION}")
target_include_directories(caelestia-core PRIVATE "${CAELESTIA_SDK}/plugin/src/Caelestia")
add_executable(syncshell-caelestia-probe probe.cpp)
target_include_directories(syncshell-caelestia-probe PRIVATE "${CAELESTIA_SDK}/plugin/src/Caelestia/Plugins")
target_link_libraries(syncshell-caelestia-probe PRIVATE caelestia-plugins Qt::Core Qt::Qml Qt::Quick)
add_executable(syncshell-caelestia-loader loader.cpp)
target_include_directories(syncshell-caelestia-loader PRIVATE "${CAELESTIA_SDK}/plugin/src/Caelestia/Plugins")
target_link_libraries(syncshell-caelestia-loader PRIVATE caelestia-plugins Qt::Core Qt::Qml Qt::Quick)
